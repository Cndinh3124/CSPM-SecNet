# Add to backend/app/main.py only when you are ready to run the React dev server.
# This is required because React (localhost:5173) and FastAPI (localhost:8000)
# are different origins.

from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
